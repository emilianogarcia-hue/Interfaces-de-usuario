final List<http.Request> requests = <http.Request>[];
http.Response Function(http.Request request) server = (_) =>
    http.Response('{}', 200);

http.Response json(Object body, [int status = 200]) => http.Response(
  jsonEncode(body),
  status,
  headers: <String, String>{'content-type': 'application/json; charset=utf-8'},
);

Map<String, dynamic> user(String email) => <String, dynamic>{
  'id': '6f1c2b1e-0000-4000-8000-000000000001',
  'aud': 'authenticated',
  'role': 'authenticated',
  'email': email,
  'app_metadata': <String, dynamic>{},
  'user_metadata': <String, dynamic>{
    'full_name': 'Ana Pérez',
    'colony': 'Santa Fe',
  },
  'created_at': '2026-10-01T00:00:00Z',
};

Map<String, dynamic> session(String email) => <String, dynamic>{
  'access_token': 'token-de-prueba',
  'token_type': 'bearer',
  'expires_in': 3600,
  'expires_at': 4102444800,
  'refresh_token': 'refresh-de-prueba',
  'user': user(email),
};

/// Respuestas comunes de la API REST (perfil y listas vacías).
http.Response rest(http.Request request) {
  final String path = request.url.path;
  if (path.endsWith('/rest/v1/profiles')) {
    return json(<String, dynamic>{
      'full_name': 'Ana Pérez',
      'colony': 'Santa Fe',
      'reports_count': 2,
      'recycled_kg': 0,
    });
  }
  if (path.contains('/rest/v1/')) {
    return json(<dynamic>[]);
  }
  return json(<String, dynamic>{});
}

/// Servidor que acepta el registro y el inicio de sesión de Ana.
http.Response okServer(http.Request request) {
  final String path = request.url.path;
  if (path.endsWith('/auth/v1/signup')) {
    final Map<String, dynamic> body =
        jsonDecode(request.body) as Map<String, dynamic>;
    return json(session(body['email'] as String));
  }
  if (path.endsWith('/auth/v1/token')) {
    final Map<String, dynamic> body =
        jsonDecode(request.body) as Map<String, dynamic>;
    if (body['email'] == 'ana@correo.mx' && body['password'] == 'Clave2026') {
      return json(session('ana@correo.mx'));
    }
    return json(<String, dynamic>{
      'code': 400,
      'error_code': 'invalid_credentials',
      'msg': 'Invalid login credentials',
    }, 400);
  }
  if (path.endsWith('/auth/v1/recover')) {
    return json(<String, dynamic>{});
  }
  if (path.endsWith('/auth/v1/verify')) {
    final String? token =
        (jsonDecode(request.body) as Map<String, dynamic>)['token'] as String?;
    if (token != '123456') {
      return json(<String, dynamic>{
        'code': 403,
        'error_code': 'otp_expired',
        'msg': 'Token has expired or is invalid',
      }, 403);
    }
    return json(session('ana@correo.mx'));
  }
  if (path.endsWith('/auth/v1/user')) {
    final Map<String, dynamic> body = request.body.isEmpty
        ? <String, dynamic>{}
        : jsonDecode(request.body) as Map<String, dynamic>;
    if (body['password'] == 'Clave2026') {
      // Respuesta real de Supabase Auth cuando la nueva contraseña es igual.
      return json(<String, dynamic>{
        'code': 422,
        'error_code': 'same_password',
        'msg': 'New password should be different from the old password.',
      }, 422);
    }
    return json(user('ana@correo.mx'));
  }
  return rest(request);
}

Iterable<http.Request> calls(String suffix) =>
    requests.where((r) => r.url.path.endsWith(suffix));

Future<void> network(WidgetTester tester) async {
  for (int i = 0; i < 20; i++) {
    await tester.runAsync(
      () => Future<void>.delayed(const Duration(milliseconds: 20)),
    );
    await tester.pump(const Duration(milliseconds: 50));
  }
  await tester.pumpAndSettle();
}

/// Cierra la app completa y el canal en tiempo real que abre Inicio.
Future<void> closeApp(WidgetTester tester) async {
  await tester.pumpWidget(const SizedBox());
  await Supabase.instance.client.removeAllChannels();
  await tester.runAsync(() => Supabase.instance.client.realtime.disconnect());
  await tester.pump(const Duration(seconds: 30));
}

Future<void> shot(String id) async {
  FocusManager.instance.primaryFocus?.unfocus();
  await expectLater(
    find.byType(MaterialApp).first,
    matchesGoldenFile('out2/$id.png'),
  );
}

/// Abre "Crear cuenta" desde el inicio de sesión, como en la app.

Future<void> fillRegister(
  WidgetTester tester, {
  String name = 'Ana Pérez',
  String email = 'ana@correo.mx',
  String password = 'Clave2026',
  String confirm = 'Clave2026',
  String? colony = 'Santa Fe',
  bool terms = true,
}) async {
  final Finder fields = find.byType(TextFormField);
  await tester.enterText(fields.at(0), name);
  await tester.enterText(fields.at(1), email);
  await tester.enterText(fields.at(2), password);
  await tester.enterText(fields.at(3), confirm);
  if (colony != null) {
    await tester.tap(find.byType(DropdownButtonFormField<String>));
    await tester.pumpAndSettle();
    await tester.tap(find.text(colony).last);
    await tester.pumpAndSettle();
  }
  if (terms) {
    await tester.tap(find.byType(Checkbox));
    await tester.pumpAndSettle();
  }
}

Future<void> tapCreate(WidgetTester tester) async {
  await tester.tap(find.widgetWithText(FilledButton, 'Crear cuenta'));
  await network(tester);
}

/// Registro completo correcto: cuenta creada y regreso al inicio de sesión.
Future<void> registerOk(
  WidgetTester tester,
  String id, {
  String email = 'ana@correo.mx',
}) async {
  server = okServer;
  await openRegister(tester);
  await fillRegister(tester, email: email);
  await shot('${id}_antes');
  await tapCreate(tester);
  final List<http.Request> signups = calls('/signup').toList();
  expect(signups, hasLength(1));
  final Map<String, dynamic> body =
      jsonDecode(signups.single.body) as Map<String, dynamic>;
  expect(body['email'], email);
  expect(body['password'], 'Clave2026');
  expect((body['data'] as Map<String, dynamic>)['full_name'], 'Ana Pérez');
  expect((body['data'] as Map<String, dynamic>)['colony'], 'Santa Fe');
  expect(
    find.text('Cuenta creada correctamente. Ya puedes iniciar sesión.'),
    findsOneWidget,
  );
  expect(find.byType(RegisterScreen), findsNothing);
  expect(find.text('Inicia sesión para continuar'), findsOneWidget);
  expect(Supabase.instance.client.auth.currentSession, isNull);
}
