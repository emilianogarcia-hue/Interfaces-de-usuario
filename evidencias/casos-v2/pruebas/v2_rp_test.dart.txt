Map<String, dynamic> center(
  int id,
  String name,
  String address,
  String colony,
  double lat,
  double lng,
  String phone,
  String hours,
  bool open,
  double rating,
  List<String> materials,
) => <String, dynamic>{
  'id': id,
  'slug': 'c$id',
  'name': name,
  'address': address,
  'colony': colony,
  'latitude': lat,
  'longitude': lng,
  'phone': phone,
  'opening_hours': hours,
  'open_now': open,
  'rating': rating,

      .map(
        (m) => <String, dynamic>{
          'recycling_materials': <String, dynamic>{'name': m},
        },
      )
      .toList(),
};

/// GPS simulado: [answers] son las respuestas al pedir permiso, en orden.
class Geo extends GeolocatorPlatform with MockPlatformInterfaceMixin {
  Geo(this.answers);

  final List<LocationPermission> answers;
  int requests = 0;

  @override
  Future<bool> isLocationServiceEnabled() async => true;

  @override
  Future<LocationPermission> checkPermission() async =>
      requests == 0 ? LocationPermission.denied : _last;

  LocationPermission _last = LocationPermission.denied;

  @override
  Future<LocationPermission> requestPermission() async {
    final LocationPermission answer =
        answers[requests < answers.length ? requests : answers.length - 1];
    requests++;
    _last = answer;
    return answer;
  }

  @override
  Future<Position> getCurrentPosition({
    LocationSettings? locationSettings,
  }) async => Position(
    latitude: 19.3653,
    longitude: -99.2900,
    timestamp: DateTime(2026, 10, 7),
    accuracy: 5,
    altitude: 0,
    altitudeAccuracy: 0,
    heading: 0,
    headingAccuracy: 0,
    speed: 0,
    speedAccuracy: 0,
  );
}

final Geo denied = Geo(<LocationPermission>[LocationPermission.denied]);

Future<void> openCenters(
  WidgetTester tester, {
  Geo? geo,
  double height = 3400,
}) async {
  GeolocatorPlatform.instance =
      geo ?? Geo(<LocationPermission>[LocationPermission.denied]);
  server = (request) => request.url.path.endsWith('/recycling_centers')
      ? json(centersData)
      : json(<String, dynamic>{});
  phone(tester, height: height);
  await tester.pumpWidget(
    MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: theme,
      home: const RecyclingCentersScreen(),
    ),
  );
  await network(tester);
}

/// Nombres de centros visibles, de arriba hacia abajo.
List<String> order(WidgetTester tester) {
  final List<String> visible = allNames
      .where((n) => find.text(n).evaluate().isNotEmpty)
      .toList();
  visible.sort(
    (a, b) => tester
        .getTopLeft(find.text(a).first)
        .dy
        .compareTo(tester.getTopLeft(find.text(b).first).dy),
  );
  return visible;
}

/// El panel "Filtros" dispara en modo debug un aviso de Flutter
/// ("ListTile background color or ink splashes may be invisible") que no se
/// ve en la app; se ignora solo ese aviso.
Future<void> ignoringListTileWarning(Future<void> Function() body) async {
  final void Function(FlutterErrorDetails)? original = FlutterError.onError;
  FlutterError.onError = (details) {
    if (details.exception.toString().contains('ListTile background color')) {
      return;
    }
    original?.call(details);
  };
  try {
    await body();
  } finally {
    FlutterError.onError = original;
  }
}

/// Repositorio simulado que aplica los cambios al perfil guardado.
class SavingRepo extends FakeProfileRepository {
  SavingRepo(super.profile, {this.failSave = false, this.session});

  final bool failSave;
  final ValueNotifier<bool>? session;

  @override
  Future<void> update(ProfileUpdate update) async {
    if (failSave) {
      throw Exception('sin conexión');
    }
    await super.update(update);
    profile = UserProfile(
      fullName: update.fullName.trim(),
      email: profile.email,
      colony: update.colony,
      reportsCount: profile.reportsCount,
      recycledKg: profile.recycledKg,
      phone: update.phone,
      bio: update.bio,
      notifications: update.notifications,
    );
  }

  @override
  Future<void> deleteAccount() async {
    await super.deleteAccount();
    session?.value = false;
  }
}
