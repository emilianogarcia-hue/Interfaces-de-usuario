/// Repositorio de campañas cuya red falla al inscribirse o cancelar.
class OfflineCampaignsRepository extends FakeCampaignsRepository {
  OfflineCampaignsRepository(super.campaigns);

  @override
  Future<void> join(int campaignId) async => throw Exception('sin conexión');

  @override
  Future<void> leave(int campaignId) async => throw Exception('sin conexión');
}

/// Notificaciones simuladas con contador de no leídas fijo.
class CountingNotificationsRepository extends FakeNotificationsRepository {
  CountingNotificationsRepository(super.items, {this.failDelete = false});

  final bool failDelete;

  @override
  Stream<int> unreadCount() =>
      Stream<int>.value(items.where((n) => !n.isRead).length);

  @override
  Future<void> delete(int id) async {
    if (failDelete) {
      // La petición sin red tarda un momento en fallar.
      await Future<void>.delayed(const Duration(milliseconds: 300));
      throw Exception('sin conexión');
    }
    return super.delete(id);
  }
}

/// Mismo armado que MainShell (pestañas en IndexedStack + barra inferior),
/// pero con las pantallas creadas con repositorios simulados.
class TestShell extends StatefulWidget {
  const TestShell({super.key, required this.tabs, this.initial = AppTab.inicio});

  final Map<AppTab, Widget> tabs;
  final AppTab initial;

  @override
  State<TestShell> createState() => _TestShellState();
}

class _TestShellState extends State<TestShell> {
  late AppTab _current = widget.initial;
  final ValueNotifier<int> _version = ValueNotifier<int>(0);

  void _select(AppTab tab) => setState(() => _current = tab);

  @override
  Widget build(BuildContext context) {
    return MainShellScope(
      currentTab: _current,
      selectTab: _select,
      openReport: () async {},
      notifyDataChanged: () => _version.value++,
      dataVersion: _version,
      child: Scaffold(
        body: IndexedStack(
          index: AppTab.values.indexOf(_current),
          children: AppTab.values
              .map((tab) => widget.tabs[tab] ?? const SizedBox.shrink())
              .toList(),
        ),
        bottomNavigationBar: AppBottomNav(
          current: _current,
          onSelected: _select,
          onReport: () {},
        ),
      ),
    );
  }
}

AppNotification notif(
  int id,
  NotificationType type,
  String title,
  String body, {
  Duration ago = const Duration(minutes: 5),
  bool read = false,
}) {
  final DateTime now = DateTime.now();
  return AppNotification(
    id: id,
    type: type,
    title: title,
    body: body,
    createdAt: now.subtract(ago),
    readAt: read ? now.subtract(ago).add(const Duration(minutes: 1)) : null,
  );
}

Finder cardOf(String title) => find.ancestor(
  of: find.text(title),
  matching: find.byType(AnimatedContainer),
);

double arrowTurns(String title) => (find
            .descendant(of: cardOf(title), matching: find.byType(AnimatedRotation))
            .evaluate()
            .first
            .widget
        as AnimatedRotation)
    .turns;

final Finder badge = find.byWidgetPredicate(
  (w) =>
      w is Container &&
      w.decoration is BoxDecoration &&
      (w.decoration! as BoxDecoration).color == const Color(0xFFFFC857),
);

        Iterable<Map<String, dynamic>> result = rows;

          result = rows.where(
