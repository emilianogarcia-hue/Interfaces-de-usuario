  jsonEncode(body),

Map<String, dynamic> row(
  int id,
  String userId,
  String category,
  String colony,
  String description, {
  String status = 'pendiente',
  String createdAt = '2026-10-05T15:00:00Z',
}) => <String, dynamic>{
  'id': id,
  'user_id': userId,
  'folio': 'ECO-2026-${id.toString().padLeft(4, '0')}',
  'category': category,
  'colony': colony,
  'description': description,
  'status': status,
  'created_at': createdAt,
  'latitude': null,
  'longitude': null,
  'photo_url': null,
};

/// Tabla `reports` simulada que respeta el filtro user_id=eq.<id>.
List<Map<String, dynamic>> table = <Map<String, dynamic>>[];

http.Response reportsServer(http.Request request) {
  if (!request.url.path.endsWith('/rest/v1/reports')) {
    return json(<String, dynamic>{});
  }
  if (request.method == 'POST') {
    final Map<String, dynamic> body =
        jsonDecode(request.body) as Map<String, dynamic>;
    final int id = 42;
    final Map<String, dynamic> inserted = row(
      id,
      body['user_id'] as String,
      body['category'] as String,
      body['colony'] as String,
      body['description'] as String,
      createdAt: DateTime.now().toUtc().toIso8601String(),
    );
    table.insert(0, inserted);
    return json(inserted, 201);
  }
  final String? filter = request.url.queryParameters['user_id'];
  final List<Map<String, dynamic>> rows = table
      .where((r) => filter == null || filter == 'eq.${r['user_id']}')
      .toList();
  return json(rows);
}

Future<void> network(WidgetTester tester) async {
  for (int i = 0; i < 20; i++) {
    await tester.runAsync(
      () => Future<void>.delayed(const Duration(milliseconds: 20)),
    );
    await tester.pump(const Duration(milliseconds: 50));
  }
  await tester.pumpAndSettle();
}

class CountingLocationService extends LocationService {
  CountingLocationService();
  int calls = 0;

  @override
  Future<GeoPoint> getCurrentLocation() async {
    calls++;
    return const GeoPoint(19.36721, -99.29214);
  }
}

/// GPS encendido; el permiso empieza sin otorgar y el usuario lo acepta.
class AcceptingGeolocator extends GeolocatorPlatform
    with MockPlatformInterfaceMixin {
  int permissionRequests = 0;

  @override
  Future<bool> isLocationServiceEnabled() async => true;

  @override
  Future<LocationPermission> checkPermission() async =>
      LocationPermission.denied;

  @override
  Future<LocationPermission> requestPermission() async {
    permissionRequests++;
    return LocationPermission.whileInUse;
  }

  @override
  Future<Position> getCurrentPosition({
    LocationSettings? locationSettings,
  }) async => Position(
    latitude: 19.36721,
    longitude: -99.29214,
    timestamp: DateTime(2026, 10, 9),
    accuracy: 5,
    altitude: 0,
    altitudeAccuracy: 0,
    heading: 0,
    headingAccuracy: 0,
    speed: 0,
    speedAccuracy: 0,
  );
}

/// Repositorio cuya recarga queda pendiente hasta completar [next].
class SlowReportsRepository implements ReportsRepository {
  SlowReportsRepository(this.first);

  final List<Report> first;
  Completer<List<Report>>? next;
  int calls = 0;

  @override
  Future<List<Report>> myReports({int? limit}) {
    calls++;
    if (calls == 1) return Future<List<Report>>.value(first);
    next = Completer<List<Report>>();
    return next!.future;
  }

  @override
  Future<Report> submit(NewReport report) => throw UnimplementedError();
}
